建 `backend/app/db/models.py`：

```python
from datetime import datetime
from sqlalchemy import String, Integer, Float, Boolean, DateTime, ForeignKey, Text
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column

class Base(DeclarativeBase):
    pass

class Product(Base):
    __tablename__ = "products"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(200))
    category: Mapped[str] = mapped_column(String(50), index=True)
    price: Mapped[float] = mapped_column(Float)
    cost: Mapped[float] = mapped_column(Float)
    created_at: Mapped[datetime] = mapped_column(DateTime)
    is_active: Mapped[bool] = mapped_column(Boolean, default=True, index=True)

class Order(Base):
    __tablename__ = "orders"
    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(Integer, index=True)
    order_date: Mapped[datetime] = mapped_column(DateTime, index=True)
    status: Mapped[str] = mapped_column(String(20))
    total_amount: Mapped[float] = mapped_column(Float)
    channel: Mapped[str] = mapped_column(String(20))

class OrderItem(Base):
    __tablename__ = "order_items"
    id: Mapped[int] = mapped_column(primary_key=True)
    order_id: Mapped[int] = mapped_column(ForeignKey("orders.id"), index=True)
    product_id: Mapped[int] = mapped_column(ForeignKey("products.id"), index=True)
    quantity: Mapped[int] = mapped_column(Integer)
    unit_price: Mapped[float] = mapped_column(Float)
    refunded: Mapped[bool] = mapped_column(Boolean, default=False, index=True)

class User(Base):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(primary_key=True)
    region: Mapped[str] = mapped_column(String(20), index=True)
    register_date: Mapped[datetime] = mapped_column(DateTime)

class OrdersArchive(Base):
    """归档表：故意留作 Schema 路由的干扰项，不应被问数选中。"""
    __tablename__ = "orders_archive"
    id: Mapped[int] = mapped_column(primary_key=True)
    user_id: Mapped[int] = mapped_column(Integer)
    order_date: Mapped[datetime] = mapped_column(DateTime)
    status: Mapped[str] = mapped_column(String(20))
    total_amount: Mapped[float] = mapped_column(Float)
    channel: Mapped[str] = mapped_column(String(20))

class TableMetadata(Base):
    """Schema 路由的元数据表。"""
    __tablename__ = "table_metadata"
    id: Mapped[int] = mapped_column(primary_key=True)
    table_name: Mapped[str] = mapped_column(String(50), unique=True)
    description: Mapped[str] = mapped_column(Text)
    key_fields: Mapped[str] = mapped_column(Text)  # JSON 字符串
```

建 `backend/app/db/seed.py`：

```python
"""
ShopLens 种子数据脚本。

故意制造的业务"坑"（用于验证 Agent 鲁棒性）：
1. "销量"有歧义：order_items.quantity 是件数，orders.total_amount 是金额
2. 8% 的 order_items.refunded = 1：销量是否排除退款
3. 10% 的 products.is_active = 0：下架商品不应出现在排行
4. 存在 orders_archive：Schema 路由的干扰项
5. 时间相对 now() 生成：保证"最近 7 天"永远有数据
"""
import json
import random
from datetime import datetime, timedelta

from faker import Faker
from app.db.base import engine, SessionLocal
from app.db.models import (
    Base, Product, Order, OrderItem, User, OrdersArchive, TableMetadata,
)

fake = Faker("zh_CN")
random.seed(42)  # 可复现

CATEGORIES = ["美妆", "数码", "家居", "服饰", "食品", "母婴", "运动", "图书", "宠物", "汽车用品"]
PRODUCT_SUFFIX = ["水乳", "耳机", "收纳盒", "连衣裙", "零食礼包", "奶瓶", "瑜伽垫", "小说", "猫粮", "车载支架"]
REGIONS = ["广东", "浙江", "江苏", "上海", "北京", "四川", "山东", "福建"]
CHANNELS = ["App", "小程序", "Web", "线下"]


def seed() -> None:
    Base.metadata.drop_all(engine)
    Base.metadata.create_all(engine)
    db = SessionLocal()
    now = datetime.now()
    try:
        # users
        db.bulk_save_objects([
            User(id=i + 1, region=random.choice(REGIONS),
                 register_date=now - timedelta(days=random.randint(30, 500)))
            for i in range(500)
        ])

        # products（10% 下架）
        db.bulk_save_objects([
            Product(
                id=i + 1,
                name=f"{random.choice(['轻奢','经典','新款','限定'])}{fake.word()}{random.choice(PRODUCT_SUFFIX)}",
                category=random.choice(CATEGORIES),
                price=(p := round(random.uniform(10, 800), 2)),
                cost=round(p * random.uniform(0.3, 0.7), 2),
                created_at=now - timedelta(days=random.randint(10, 400)),
                is_active=random.random() > 0.1,
            )
            for i in range(300)
        ])

        # orders（90 天跨度）+ order_items（每单 1~5 条，8% 退款）
        orders, items, item_id = [], [], 1
        for i in range(8000):
            days_ago = random.randint(0, 89)
            oid = i + 1
            total = 0.0
            orders.append(Order(
                id=oid, user_id=random.randint(1, 500),
                order_date=now - timedelta(days=days_ago, hours=random.randint(0, 23)),
                status=random.choice(["paid", "paid", "paid", "shipped", "refunded"]),
                total_amount=0.0,  # 稍后回填
                channel=random.choice(CHANNELS),
            ))
            for _ in range(random.randint(1, 5)):
                pid = random.randint(1, 300)
                qty = random.choice([1, 1, 1, 2, 2, 3, 4, 5])
                unit = round(random.uniform(10, 800), 2)
                total += qty * unit
                items.append(OrderItem(
                    id=item_id, order_id=oid, product_id=pid,
                    quantity=qty, unit_price=unit,
                    refunded=random.random() < 0.08,
                ))
                item_id += 1
            orders[-1].total_amount = round(total, 2)

        db.bulk_save_objects(orders)
        db.bulk_save_objects(items)

        # orders_archive：仅建表 + 少量历史数据，作为 Schema 路由干扰项
        db.bulk_save_objects([
            OrdersArchive(
                id=i + 1, user_id=random.randint(1, 500),
                order_date=now - timedelta(days=random.randint(365, 730)),
                status="paid", total_amount=round(random.uniform(50, 2000), 2),
                channel=random.choice(CHANNELS),
            )
            for i in range(200)
        ])

        # table_metadata（Schema 路由的种子）
        meta = [
            ("products", "商品表，包含商品名称、品类、价格、成本、是否上架。销量排行时必须过滤 is_active=1。",
             ["name", "category", "price", "is_active"]),
            ("orders", "订单主表，包含下单时间、订单状态、总金额、渠道。total_amount 是金额不是件数。",
             ["order_date", "status", "total_amount", "channel"]),
            ("order_items", "订单明细表，quantity 是件数，refunded=1 表示该明细已退款。计算销量时应排除退款。",
             ["product_id", "quantity", "unit_price", "refunded"]),
            ("users", "用户表，包含注册地区和注册时间。",
             ["region", "register_date"]),
            ("orders_archive", "历史归档订单表（一年前），不用于近期经营分析。",
             ["order_date", "total_amount"]),
        ]
        db.bulk_save_objects([
            TableMetadata(table_name=n, description=d, key_fields=json.dumps(f, ensure_ascii=False))
            for n, d, f in meta
        ])

        db.commit()
        print(f"✅ seed done: users=500 products=300 orders=8000 items={len(items)}")
    except Exception:
        db.rollback()
        raise
    finally:
        db.close()


if __name__ == "__main__":
    seed()
```

## Step 4：用硬编码 SQL 验证数据能支撑问数场景

建 `backend/scripts/check_data.py`：

```python
"""验证数据能回答核心问数场景：近 7 天销量 TOP5（排除退款、排除下架商品）。"""
from sqlalchemy import text
from app.db.base import SessionLocal

SQL = """
SELECT p.name, p.category, SUM(oi.quantity) AS sales
FROM order_items oi
JOIN orders o        ON oi.order_id = o.id
JOIN products p      ON oi.product_id = p.id
WHERE o.order_date >= datetime('now', '-7 days')
  AND oi.refunded = 0
  AND p.is_active = 1
GROUP BY p.id
ORDER BY sales DESC
LIMIT 5;
"""

def main() -> None:
    db = SessionLocal()
    try:
        rows = db.execute(text(SQL)).fetchall()
        print("近 7 天销量 TOP5（已排除退款与下架）：")
        for i, (name, cat, sales) in enumerate(rows, 1):
            print(f"  {i}. {name} [{cat}] — {sales} 件")
        assert len(rows) == 5, "结果不足 5 条，检查数据分布"
        print("✅ check passed")
    finally:
        db.close()

if __name__ == "__main__":
    main()
```

运行：

```bash
python -m scripts.check_data
```

**注意 `scripts/` 下要有 `__init__.py`**（`touch backend/scripts/__init__.py`），否则 `python -m scripts.check_data` 会找不到模块。

看到 5 行排行 + `✅ check passed`，说明数据模型和问数场景是对齐的。这条 SQL 就是 P1 阶段“硬编码端到端”要用的那条。

## Step 5：写 SSE 事件协议文档

建 `docs/sse-protocol.md`：

```markdown
# ShopLens SSE 事件协议 v0.1

所有事件通过 `text/event-stream` 传输，每条事件的 `data` 为 JSON 字符串。
前端按 `event` 字段分发，**只增不改**：新增事件类型向下兼容，已有字段不删不改语义。

## 事件类型

| event | 触发时机 | data 字段 | 前端行为 |
|---|---|---|---|
| `stage` | Agent 进入某阶段 | `{ "stage": "routing"\|"generating"\|"validating"\|"executing"\|"charting"\|"rewriting" }` | 更新状态条文字 |
| `token` | 模型流式输出文字 | `{ "text": "..." }` | 追加到消息气泡（打字机） |
| `sql` | 生成/重写了 SQL | `{ "sql": "SELECT ...", "version": 1 }` | 存入消息对象，Trace 面板展示，不直接渲染 |
| `chart` | 图表声明就绪 | `{ "chart_type": "bar"\|"line"\|"pie"\|"table", "x_field": "...", "y_field": "...", "title": "...", "data": [{...}] }` | 交给 `ChartBlock.vue` 渲染 |
| `trace` | 单步决策记录 | `{ "step": "...", "status": "ok"\|"error", "detail": "...", "elapsed_ms": 123 }` | 追加到 Trace 时间线 |
| `error` | 出错 | `{ "code": "sql_invalid"\|"llm_error"\|"tool_timeout", "message": "...", "recoverable": true }` | `recoverable=true` 显示“正在修复”，否则显示失败 |
| `done` | 流正常结束 | `{}` | 关闭连接，恢复输入框 |

## 示例流（一次成功问数）
```

event: stage
data: {"stage":"routing"}

event: stage
data: {"stage":"generating"}

event: sql
data: {"sql":"SELECT ...","version":1}

event: stage
data: {"stage":"validating"}

event: trace
data: {"step":"sql_guard","status":"ok","detail":"只读校验通过","elapsed_ms":12}

event: stage
data: {"stage":"executing"}

event: chart
data: {"chart_type":"bar","x_field":"name","y_field":"sales","title":"近7天销量TOP5","data":[...]}

event: token
data: {"text":"近 7 天销量最高的是..."}

event: done
data: {}

```
## 约定

- 一次请求以 `done` 或 `error(recoverable=false)` 结束，二者必居其一。
- `sql` 事件可能出现多次（重写时 `version` 递增）。
- 前端不得假设事件顺序绝对固定，但 `done` 一定在最后。
- 心跳由后端 `sse-starlette` 自动发送（15s），前端忽略注释行（`:` 开头）。
```

**几个容易踩的点提前说**：

1. `python -m scripts.check_data` 报 `No module named scripts` —— 检查 `backend/scripts/__init__.py` 是否存在，且当前目录是 `backend/`。
2. `python -m app.db.seed` 报 `ModuleNotFoundError: app` —— 必须**在 `backend/` 目录下**运行，不能在项目根目录。
3. seed 跑得慢（8000 单 + 两万多明细）属正常，`bulk_save_objects` 一般 10 秒内完成；如果超过 1 分钟，检查是不是每条都 `db.add()` 了。
4. `datetime('now', '-7 days')` 是 SQLite 语法，换 MySQL 时要改成 `DATE_SUB(NOW(), INTERVAL 7 DAY)`——这个差异记下来，P2 的 SQLGlot 方言校验会用到。
